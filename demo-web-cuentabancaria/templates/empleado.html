# EJERCICIO 2: ENCAPSULACIÓN
# Protege los atributos privados (__nombre, __edad)
# usando getters y setters con decoradores @property.
# ==========================================

class Empleado:
    # 1. Constructor: Delega la asignación a los setters para validar
    def __init__(self, nombre: str, edad: int):
        self.nombre = nombre  # Activa el setter @nombre.setter
        self.edad = edad      # Activa el setter @edad.setter

    # 2. Getter de Nombre: Lectura segura del atributo privado __nombre
    @property
    def nombre(self) -> str:
        return self.__nombre

    # 3. Setter de Nombre: Validación (no vacío ni espacios en blanco)
    @nombre.setter
    def nombre(self, valor: str) -> None:
        if not valor or not valor.strip():
            raise ValueError("El nombre no puede estar vacío")
        self.__nombre = valor

    # 4. Getter de Edad: Lectura segura del atributo privado __edad
    @property
    def edad(self) -> int:
        return self.__edad

    # 5. Setter de Edad: Validación de regla de negocio (0 < edad < 100)
    @edad.setter
    def edad(self, valor: int) -> None:
        if valor <= 0 or valor >= 100:
            raise ValueError("La edad debe ser mayor que 0 y menor que 100")
        self.__edad = valor
