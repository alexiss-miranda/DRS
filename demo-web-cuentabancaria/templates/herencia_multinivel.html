# ==========================================
# EJERCICIO 5: HERENCIA MULTINIVEL
# Jerarquía vertical de 3 niveles:
# Nivel 1 (Animal) -> Nivel 2 (Mamifero) -> Nivel 3 (Perro)
# ==========================================

# 1. Nivel 1 (Clase Raíz): Define comportamiento general
class Animal:
    def hacer_sonido(self) -> str:
        return "El animal hace un sonido genérico."


# 2. Nivel 2 (Clase Intermedia): Hereda de Animal y agrega alimentar()
class Mamifero(Animal):
    def alimentar(self) -> str:
        return "El mamífero alimenta a sus crías."


# 3. Nivel 3 (Clase Hoja Derivada): Hereda de Mamifero y sobrescribe hacer_sonido()
class Perro(Mamifero):
    # Sobrescritura de método heredado de Animal
    def hacer_sonido(self) -> str:
        return "El perro ladra: ¡Guau!"


# 4. Resolución de Métodos (MRO: Perro -> Mamifero -> Animal -> object)
perro = Perro()
perro.alimentar()    # Resuelto en Nivel 2 (Mamifero)
perro.hacer_sonido() # Resuelto en Nivel 3 (Perro)
