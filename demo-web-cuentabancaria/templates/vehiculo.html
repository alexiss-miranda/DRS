# ==========================================
# EJERCICIO 3: HERENCIA SIMPLE
# La subclase Coche reutiliza el código de la
# superclase Vehiculo y añade métodos propios.
# ==========================================

# 1. Superclase / Clase Base: Comportamiento general
class Vehiculo:
    def arrancar(self) -> str:
        return "El vehículo ha arrancado."
        
    def detener(self) -> str:
        return "El vehículo se ha detenido."


# 2. Subclase: Coche hereda arrancar() y detener()
class Coche(Vehiculo):
    # Método exclusivo de la subclase Coche
    def conducir(self) -> str:
        return "El coche está siendo conducido."


# 3. Demostración de uso:
mi_coche = Coche()
mi_coche.arrancar()  # Heredado de Vehiculo
mi_coche.conducir()   # Propio de Coche
mi_coche.detener()    # Heredado de Vehiculo
