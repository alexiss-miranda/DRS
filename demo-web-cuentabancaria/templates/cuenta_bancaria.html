# EJERCICIO 1: ABSTRACCIÓN
# Oculta la complejidad interna y expone solo
# las operaciones esenciales al usuario.
# ==========================================

class CuentaBancaria:
    # 1. Constructor: Inicializa la cuenta con un saldo base
    def __init__(self, saldo_inicial: Decimal = Decimal("0.00")):
        self._saldo = saldo_inicial  # Estado interno encapsulado

    # 2. Depositar: Valida la regla (monto > 0) y actualiza
    def depositar(self, monto: Decimal) -> None:
        if monto <= 0:
            raise ValueError("El monto a depositar debe ser positivo")
        self._saldo += monto

    # 3. Retirar: Valida que haya saldo suficiente antes de descontar
    def retirar(self, monto: Decimal) -> None:
        if monto > self._saldo:
            raise SaldoInsuficienteError("Fondos insuficientes para el retiro")
        self._saldo -= monto

    # 4. Obtener saldo: Consulta el saldo actual de forma segura
    def obtener_saldo(self) -> Decimal:
        return self._saldo
